def combinationSum3(k, n):
    result = []

    def backtrack(start, path, total):
        # If combination size becomes k
        if len(path) == k:
            if total == n:
                result.append(path[:])
            return

        # Try numbers from start to 9
        for i in range(start, 10):
            if total + i > n:
                break

            path.append(i)
            backtrack(i + 1, path, total + i)
            path.pop()

    backtrack(1, [], 0)
    return result


# Input
k = int(input())
n = int(input())

# Output
print(combinationSum3(k, n))
